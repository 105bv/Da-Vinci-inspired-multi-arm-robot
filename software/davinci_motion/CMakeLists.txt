cmake_minimum_required(VERSION 3.8)
project(davinci_motion)
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(motion_smoke_test src/motion_smoke_test.cpp)
target_include_directories(motion_smoke_test PRIVATE include)
target_compile_features(motion_smoke_test PUBLIC cxx_std_17)
ament_target_dependencies(motion_smoke_test
  rclcpp rclcpp_action control_msgs trajectory_msgs sensor_msgs std_srvs)
install(TARGETS motion_smoke_test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_arm_model test/test_arm_model.cpp)
  target_include_directories(test_arm_model PRIVATE include)
  target_compile_features(test_arm_model PUBLIC cxx_std_17)
endif()
ament_package()
